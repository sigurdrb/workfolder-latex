  \frametitle{System overview}
\begin{minipage}[t]{0.5\textwidth}
\scriptsize 
    \begin{enumerate}
    \vspace{0pt} % <-- forces top alignment inside the minipage
    \item<1-> Dead fish are gathered at the bottom of the pen.
    \item<2-> The fish flows through a GRP pipe from the pen and up to the LiftUp unit.
    \item<3-> Stunner unit (state mandatory)
    \item<4-> It then moves through the liftup-unit by means of gravity and a VSD controlled conveyor belt
    \item<5-> Flushing system
    \item<6-> Camera unit / vision system
    \item <7-> Distance sensor
    \item <8-> Moves through a pre-grinder
    \item <9-> Stored in tanks and continuously circulated/grinded in the ensilage system.
    \item <10-> Automatic handling of acid dosing
    \end{enumerate}
\end{minipage}%

\begin{frame}

  \begin{minipage}[t]{0.5\textwidth}
\scriptsize 
    \begin{enumerate}
    \vspace{0pt} % <-- forces top alignment inside the minipage
    \item<1-> Will be continuously running in subsea applications. Will only run when dead fish system is active on other pens.
    \item<2-> Responsibilities are delivering air to
    \begin{itemize}
      \item<3-> {\footnotesize a buffer tank used for refilling the air domes in the susbea pens}.

    \end{itemize}
    \item<5-> Controlled and monitored in the ensilage PLC by hardwired (+24VDC) signals.
    \item Has its own local user interface with configuration and setting possibilities.
    \end{enumerate}

\begin{frame}
  \frametitle{Ensilage unit}
  \begin{minipage}[t]{0.5\textwidth}
\scriptsize 
    \begin{enumerate}
    \vspace{0pt} % <-- forces top alignment inside the minipage
    \item<1-> Consists of two different PLC's. 
      \begin{itemize}
        \item<2-> Ensilage PLC. Responsible for reading different tank levels, valve- and pump control.
        \item<3-> Acid dosing PLC. Responsible for dosing the correct amount of acid in the grinder tank.
      \end{itemize}
    \item<4-> pre-grinder
    \begin{itemize}
      \item <4-> controlled by hardwired signals from the PLC (+24VDC).
    \end{itemize}
      \item<5-> interface to the compressor by HW signals.
      \item<6-> interface to LiftUp system over Modbus TCP.
      \item<7-> single point interface to SiteControl through OPC UA.
    \end{enumerate}
\end{minipage}%

\begin{frame}
  \frametitle{System control}
    \begin{enumerate}
    \vspace{0pt} % <-- forces top alignment inside the minipage
    \item<1-> Manual mode
    \begin{itemize}
    \item<2-> Can start each subsystems manually. 
    \item<3-> More familiarity of the process.
    \item<4-> Need to have personell on board for operating.
    \end{itemize}
    \item<5-> Semi auto mode
    \begin{itemize}
      \item<6-> The controller checks if every necessary subsystem for remote control is available.
      \item<7-> Manually select each pen. The controller takes care of starting the necessary subsystems in correct order.
    \end{itemize}
    \item<8-> Auto mode
    \begin{itemize}
      \item<9-> Similar as semi-auto, but schedule based. Starts up at regular time intervals, run each pen for a predetermined time
    \end{itemize}
    \item<10-> More manual control modes from sitecontrol?
    \end{enumerate}
\end{frame}
